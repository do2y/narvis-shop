const encoder = new TextEncoder();
const frame = (type: string, data: unknown) =>
  `data: ${JSON.stringify({ type, data })}\n\n`;
const token = (text: string) => frame("token", { text });
const DONE = frame("done", { finishReason: "stop" });

/** 모든 프레임을 담고 닫힌 SSE 응답 */
function sse(...frames: string[]): Response {
  const stream = new ReadableStream<Uint8Array>({
    start(controller) {
      for (const f of frames) controller.enqueue(encoder.encode(f));
      controller.close();
    },
  });
  return new Response(stream, { status: 200 });
}

/** 스트림 시작 전 거부 — 계약 CH-2 §실패 응답 envelope */
function rejected(status: number, code = "ERROR"): Response {
  return new Response(
    JSON.stringify({
      success: false,
      error: { code, message: "", requestId: "req-1" },
    }),
    { status },
  );
}

/**
 * 테스트가 프레임을 하나씩 밀어 넣는 스트림. abort 되면 실제 fetch 처럼
 * 읽기가 AbortError 로 끊긴다.
 */
function openStream() {
  let controller!: ReadableStreamDefaultController<Uint8Array>;
  const reply = (signal: AbortSignal) => {
    const stream = new ReadableStream<Uint8Array>({
      start(c) {
        controller = c;
      },
    });
    signal.addEventListener("abort", () =>
      controller.error(new DOMException("aborted", "AbortError")),
    );
    return new Response(stream, { status: 200 });
  };
  return {
    reply,
    push: (f: string) => controller.enqueue(encoder.encode(f)),
    close: () => controller.close(),
  };
}

// fetch 는 준비된 응답을 순서대로 하나씩 돌려준다. 준비되지 않은 호출은 곧 "나가면 안 되는 요청"이다.
type Reply = Response | ((signal: AbortSignal) => Response);
let replies: Reply[] = [];
const fetchMock = vi.fn(async (_url: string, init: RequestInit) => {
  const signal = init.signal as AbortSignal;
  if (signal.aborted) throw new DOMException("aborted", "AbortError");
  const next = replies.shift();
  if (!next) throw new Error("unexpected fetch");
  return typeof next === "function" ? next(signal) : next;
});

const obs: Record<string, unknown> = {};
afterAll(() => writeFileSync(process.env.PROBE_OUT!, JSON.stringify(obs, null, 2)));
const createDraft = { draftId: "d-create", op: "create" as const, productId: null, changes: [], summary: "" };
const updateDraft = { draftId: "d-update", op: "update" as const, productId: "7", changes: [], summary: "" };
const seed = (...drafts: (typeof createDraft | typeof updateDraft)[]) =>
  useChatStore.setState({ results: drafts.map((draft) => ({ kind: "draft" as const, draft })), messages: [{ id: "u", role: "user", text: "q" }, { id: "a", role: "assistant", text: "초안" }] });
const settledOf = () => state().results.map((r) => r.kind === "draft" ? [r.draft.draftId, r.settled?.type ?? null] : null);
describe("probe", () => {
  it("P4 일반 발화 턴에서 온 다른 상품의 성공 action", async () => {
    seed(createDraft);
    replies.push(sse(frame("action", { type: "PRODUCT_UPDATED", message: "다른 상품 수정됨", productId: "999" }), DONE));
    const { result } = setup({ channel: "SELLER" });
    await act(() => result.current.send("다른 질문"));
    obs.P4 = settledOf();
  });
  it("P5 수정 초안 승인에 다른 productId 의 action", async () => {
    seed(updateDraft);
    replies.push(sse(frame("action", { type: "PRODUCT_UPDATED", message: "수정됨", productId: "999" }), DONE));
    const { result } = setup({ channel: "SELLER" });
    await act(() => result.current.confirm("d-update"));
    obs.P5 = settledOf();
  });
  it("P6 초안 2개 중 B 승인인데 action 이 A 를 가리킴", async () => {
    seed(createDraft, updateDraft);
    replies.push(sse(frame("action", { type: "PRODUCT_UPDATED", message: "처리됨", productId: "999" }), DONE));
    const { result } = setup({ channel: "SELLER" });
    await act(() => result.current.confirm("d-update"));
    obs.P6 = settledOf();
  });
  it("P7 승인 스트림에서 성공 action 뒤 error", async () => {
    seed(createDraft);
    replies.push(sse(frame("action", { type: "PRODUCT_UPDATED", message: "등록됐어요", productId: "7" }), frame("error", { code: "INTERNAL", message: "", retryable: true })));
    const { result } = setup({ channel: "SELLER" });
    await act(() => result.current.confirm("d-create"));
    obs.P7 = { settled: settledOf(), bubble: { error: lastMessage().error, retryable: lastMessage().retryable } };
  });
  it("P8 승인 스트림이 action 없이 done{refresh} / done{keep}", async () => {
    for (const panel of ["refresh", "keep"]) {
      state().reset(); seed(createDraft);
      replies.push(sse(token("안내 문구"), frame("done", { finishReason: "stop", panel })));
      const { result, unmount } = setup({ channel: "SELLER" });
      await act(() => result.current.confirm("d-create"));
      obs["P8-" + panel] = { settled: settledOf(), error: lastMessage().error ?? null };
      unmount();
    }
  });
});
