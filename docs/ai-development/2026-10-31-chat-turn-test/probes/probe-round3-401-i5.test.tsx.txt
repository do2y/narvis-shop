import { QueryClient, QueryClientProvider } from "@tanstack/react-query";
import {
  cleanup,
  fireEvent,
  render,
  screen,
  waitFor,
  within,
} from "@testing-library/react";
import { afterEach, beforeEach, describe, expect, it, vi } from "vitest";
import { ensureSession } from "@/shared/chat/sessionCoordinator";
import { useChatStore } from "@/shared/chat/store";
import type { ChatSession, SellerDraft } from "@/shared/types/chat";
import SellerChatPage from "./ChatPage";

/**
 * 판매자 등록 카드의 "다시 시도" 연결 — 화면 전체를 렌더해 실제 배선을 지난다.
 *
 * 카드는 재시도 대상 id 를 모른다. ChatPage 가 "마지막 말풍선"으로 감싸 넘기는데,
 * 카드의 오류 상태도 그 말풍선의 error 에서 나오므로 둘이 같은 대상을 가리켜야 한다.
 * 훅 테스트(useChat.test.ts)로는 이 배선을 확인할 수 없다.
 */

vi.mock("next/navigation", () => ({
  usePathname: () => "/seller/chat",
  useSearchParams: () => new URLSearchParams(),
  useRouter: () => ({ push: vi.fn(), replace: vi.fn(), back: vi.fn() }),
}));
vi.mock("@/shared/chat/sessionCoordinator", async (importOriginal) => {
  const actual =
    await importOriginal<typeof import("@/shared/chat/sessionCoordinator")>();
  return {
    ...actual,
    ensureSession: vi.fn(),
    refreshTicket: vi.fn(),
    clearCachedSession: vi.fn(),
    subscribeSession: vi.fn(() => () => {}),
    subscribeOwnershipChange: vi.fn(() => () => {}),
  };
});
vi.mock("@/shared/analytics/track", () => ({ track: vi.fn() }));

const session: ChatSession = {
  sessionId: "session-1",
  ttlSeconds: 600,
  streamTicket: "ticket-1",
  ticketTtlSeconds: 60,
  llmSseUrl: "https://ai.example/seller/chat",
};

const encoder = new TextEncoder();
const frame = (type: string, data: unknown) =>
  `data: ${JSON.stringify({ type, data })}\n\n`;
function sse(...frames: string[]): Response {
  const stream = new ReadableStream<Uint8Array>({
    start(controller) {
      for (const f of frames) controller.enqueue(encoder.encode(f));
      controller.close();
    },
  });
  return new Response(stream, { status: 200 });
}
const rejected = (status: number) =>
  new Response(
    JSON.stringify({ success: false, error: { code: "ERROR", message: "" } }),
    { status },
  );

const draft: SellerDraft = {
  draftId: "draft-1",
  op: "create",
  productId: null,
  changes: [],
  summary: "",
  preview: {
    title: "코튼 오버핏 셔츠",
    imageUrl: null,
    imagePlaceholder: true,
    priceText: "32,000원",
    originalPriceText: null,
    discountRate: 0,
    stockText: "50개",
    categoryPath: "패션의류 > 셔츠",
    summary: null,
    description: null,
    sections: [],
  },
};

// AI 서버(SSE)로 가는 요청만 순서대로 응답한다. 그 밖의 요청(목록 조회 등)은 이 테스트와 무관해 실패로 둔다.
let replies: Response[] = [];
const aiCalls: RequestInit[] = [];
const fetchMock = vi.fn(async (url: string | URL | Request, init?: RequestInit) => {
  if (String(url) !== session.llmSseUrl) throw new Error("not under test");
  aiCalls.push(init!);
  const next = replies.shift();
  if (!next) throw new Error("unexpected AI request");
  return next;
});
const aiBody = (i: number) => JSON.parse(aiCalls[i].body as string);

function renderPage() {
  const client = new QueryClient({
    defaultOptions: { queries: { retry: false } },
  });
  return render(
    <QueryClientProvider client={client}>
      <SellerChatPage />
    </QueryClientProvider>,
  );
}

function sendFromInput(text: string) {
  fireEvent.change(screen.getByRole("textbox"), { target: { value: text } });
  fireEvent.click(screen.getByRole("button", { name: "전송" }));
}

beforeEach(() => {
  // jsdom 에는 레이아웃이 없어 scrollTo 가 구현돼 있지 않다(대화 자동 스크롤이 부른다)
  Element.prototype.scrollTo = () => {};
  useChatStore.getState().reset();
  sessionStorage.clear();
  replies = [];
  aiCalls.length = 0;
  vi.stubGlobal("fetch", fetchMock);
  vi.mocked(ensureSession).mockReset().mockResolvedValue(session);
});

afterEach(async () => {
  cleanup();
  await new Promise((r) => setTimeout(r, 0));
  vi.unstubAllGlobals();
});


// ---- 관찰용 프로브 (저장소 미반영, H1 2단계 작업 트리 기준) ----
import { writeFileSync } from "node:fs";
import { afterAll } from "vitest";
import { refreshTicket } from "@/shared/chat/sessionCoordinator";

const obs: Record<string, unknown> = {};
afterAll(() => writeFileSync(process.env.PROBE_OUT!, JSON.stringify(obs, null, 2)));

const updateDraft: SellerDraft = {
  draftId: "draft-u1",
  op: "update",
  productId: "7",
  changes: [{ field: "price", before: 32000, after: 30000 }],
  summary: "가격을 30,000원으로 인하",
};
const draftTurnOf = (d: SellerDraft) =>
  sse(
    [
      frame("meta", { lane: "product" }),
      frame("token", { text: "초안을 만들었어요." }),
      frame("draft", d),
      frame("done", { finishReason: "stop", panel: "replace" }),
    ].join(""),
  );
const lastMsg = () => useChatStore.getState().messages.slice(-1)[0];
const tick = () => new Promise((r) => setTimeout(r, 30));
const buttons = (name: RegExp) =>
  screen.queryAllByRole("button", { name }).map((b) => ({
    disabled: (b as HTMLButtonElement).disabled,
  }));

describe("probe", () => {
  it.each([
    ["create", draft, /상품 등록/],
    ["update", updateDraft, /적용/],
  ])("P1 %s 승인 — 최초 요청 401 → 자동 재발급·재전송", async (label, d, confirmBtn) => {
    vi.mocked(refreshTicket).mockReset().mockResolvedValue({ ...session, streamTicket: "ticket-2" });
    replies.push(
      draftTurnOf(d),
      rejected(401),
      sse(frame("done", { finishReason: "stop", panel: "refresh" })),
    );
    renderPage();
    sendFromInput("초안 만들어 줘");
    await screen.findAllByRole("button", { name: confirmBtn });
    fireEvent.click(screen.getAllByRole("button", { name: confirmBtn })[0]);
    await waitFor(() => expect(useChatStore.getState().isStreaming).toBe(false));
    await tick();
    obs[`P1-${label}`] = {
      aiRequests: aiCalls.length,
      confirmBodies: aiCalls.slice(1).map((c, i) => ({
        body: aiBody(i + 1),
        auth: (c.headers as Record<string, string>).Authorization,
      })),
      refreshCalls: vi.mocked(refreshTicket).mock.calls.length,
      lastMessage: { text: lastMsg().text, error: lastMsg().error },
    };
  });

  it.each([
    ["500", () => rejected(500)],
    ["끊김", () => sse(frame("token", { text: "처리 중" }))],
    ["SSE error", () => sse(frame("error", { code: "INTERNAL", message: "", retryable: true }))],
  ])("P2 수정 승인(DiffCard) 요청 후 실패(%s) → 적용 버튼 상태 · 다시 누르면?", async (label, failure) => {
    replies.push(
      draftTurnOf(updateDraft),
      failure(),
      sse(frame("done", { finishReason: "stop", panel: "refresh" })),
    );
    renderPage();
    sendFromInput("가격 내려 줘");
    await screen.findAllByRole("button", { name: /적용/ });
    fireEvent.click(screen.getAllByRole("button", { name: /적용/ })[0]);
    await waitFor(() => expect(lastMsg().error).toBeDefined());
    await waitFor(() => expect(useChatStore.getState().isStreaming).toBe(false));
    await tick();
    const afterFailure = {
      bubble: { error: lastMsg().error, retryable: lastMsg().retryable, resend: lastMsg().resend },
      applyButtons: buttons(/적용/),
      retryButtons: buttons(/다시 시도/),
    };
    let resent: unknown = null;
    const apply = screen.queryAllByRole("button", { name: /적용/ })[0];
    if (apply && !(apply as HTMLButtonElement).disabled) {
      fireEvent.click(apply);
      await waitFor(() => expect(useChatStore.getState().isStreaming).toBe(false));
      await tick();
      resent = aiCalls.length > 2 ? aiBody(2) : "요청 없음";
    }
    obs[`P2-${label}`] = { afterFailure, clickApplyAgain: resent, aiRequests: aiCalls.length };
  });

  it("P3 등록 승인(CreateCard) 요청 후 실패(500) → 등록 버튼 상태(비교)", async () => {
    replies.push(draftTurnOf(draft), rejected(500));
    renderPage();
    sendFromInput("등록해 줘");
    await screen.findByText("코튼 오버핏 셔츠");
    fireEvent.click(screen.getByRole("button", { name: /상품 등록/ }));
    await waitFor(() => expect(lastMsg().error).toBeDefined());
    await tick();
    obs["P3-create-500"] = {
      registerButtons: buttons(/상품 등록/),
      retryButtons: buttons(/다시 시도/),
      closeButtons: buttons(/초안 닫기/),
    };
  });
});
